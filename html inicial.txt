                icon = "fa-smog text-slate-300";
            } else if ([51, 53, 55, 61, 63, 80].includes(code)) {
                cond = "Chuva Leve / Garoa";
                fator = 1.30;
                icon = "fa-cloud-rain text-blue-400";
            } else {
                cond = "Chuva Forte / Tempestade";
                fator = 1.80;
                icon = "fa-cloud-showers-heavy text-indigo-400";
            }
            
            climaAtualDados = {
                cidade: cidadeNome,
                temp: Math.round(current.temperature_2m),
                umidade: current.relative_humidity_2m,
                vento: Math.round(current.wind_speed_10m),
                precipitacao: current.precipitation,
                codigo: code,
                condicao: cond,
                fatorImpacto: fator,
                iconClass: icon
            };
            
            atualizarWidgetClimaUI();
            calcularEstimativaConclusaoClima();
            mostrarToast(`Clima via API (${cidadeNome}): ${climaAtualDados.temp}°C, ${cond}`, "success");
        } catch (err) {
            console.warn("Utilizando dados simulados de clima:", err);

            calcularEstimativaConclusaoClima();
        }
    }

    function calcularEstimativaConclusaoClima() {
        if (!document.getElementById("inputVolTotal")) return;
        
        const volumeTotal = parseFloat(document.getElementById("inputVolTotal").value) || 10000;
        const volumeExecutado = parseFloat(document.getElementById("inputVolExecutado").value) || 4500;
        const ritmoDiario = parseFloat(document.getElementById("inputRitmoDiario").value) || 500;
        const fatorClimaManual = parseFloat(document.getElementById("inputFatorClimaSelect").value) || climaAtualDados.fatorImpacto;

        const restante = Math.max(0, volumeTotal - volumeExecutado);
        const diasBase = Math.ceil(restante / (ritmoDiario > 0 ? ritmoDiario : 1));
        const diasAjustados = Math.ceil(diasBase * fatorClimaManual);
        const diasAtraso = diasAjustados - diasBase;

        const hoje = new Date();
        const dataNormal = new Date(hoje);
        dataNormal.setDate(hoje.getDate() + diasBase);

        const dataAjustada = new Date(hoje);
        dataAjustada.setDate(hoje.getDate() + diasAjustados);

        const formatar = d => d.toLocaleDateString("pt-BR", { day: "2-digit", month: "2-digit", year: "numeric" });

        document.getElementById("resDiasBase").textContent = diasBase + " dias úteis";
        document.getElementById("resDiasAjustados").textContent = diasAjustados + " dias úteis (" + formatar(dataAjustada) + ")";

        const containerAtraso = document.getElementById("boxResultadoImpactoClima");
        if (diasAtraso > 0) {
            containerAtraso.className = "p-4 rounded-xl bg-amber-50 border border-amber-200 text-amber-900 text-xs space-y-1";
            containerAtraso.innerHTML = `
                <div class="flex items-center gap-2 font-bold text-amber-900">
                    <i class="fa-solid fa-triangle-exclamation text-amber-600 text-sm"></i>
                    <span>Impacto Climático Detectado: +${diasAtraso} dia(s) adicionais de execução</span>
                </div>
                <p class="mt-1">Devido ao clima desfavorável (fator ${fatorClimaManual.toFixed(2)}x), a velocidade de compactação e cura do asfalto diminui. A data estimada de conclusão da obra passa de <strong>${formatar(dataNormal)}</strong> para <strong>${formatar(dataAjustada)}</strong>.</p>
            `;
        } else {
            containerAtraso.className = "p-4 rounded-xl bg-emerald-50 border border-emerald-200 text-emerald-900 text-xs space-y-1";
            containerAtraso.innerHTML = `
                <div class="flex items-center gap-2 font-bold text-emerald-900">
                    <i class="fa-solid fa-circle-check text-emerald-600 text-sm"></i>
                    <span>Tempo Favorável: Cronograma Normal Mantido</span>
                </div>
                <p class="mt-1">Condições climáticas sem interferências severas. Data prevista de término: <strong>${formatar(dataNormal)}</strong>.</p>
            `;
        }
    }

    function verificarForcaSenha(senha) {
        const label = document.getElementById("labelForcaSenha");
        const bar = document.getElementById("barForcaSenha");
        if (!label || !bar) return;

        if (!senha) {
            label.textContent = "Insira a senha";
            label.className = "font-bold text-slate-400";
            bar.style.width = "0%";
            bar.className = "h-full w-0 bg-slate-400 transition-all duration-300";
            return;
        }

        let forca = 0;
        if (senha.length >= 6) forca += 25;
        if (senha.length >= 8) forca += 25;
        if (/[0-9]/.test(senha)) forca += 25;
        if (/[^A-Za-z0-9]/.test(senha) || /[A-Z]/.test(senha)) forca += 25;

        if (forca <= 25) {
            label.textContent = "Fraca";
            label.className = "font-bold text-red-500";
            bar.style.width = "25%";
            bar.className = "h-full w-1/4 bg-red-500 transition-all duration-300";
        } else if (forca <= 75) {
            label.textContent = "Média";
            label.className = "font-bold text-amber-500";
            bar.style.width = "65%";
            bar.className = "h-full w-2/3 bg-amber-500 transition-all duration-300";
        } else {
            label.textContent = "Forte";
            label.className = "font-bold text-emerald-500";
            bar.style.width = "100%";
            bar.className = "h-full w-full bg-emerald-500 transition-all duration-300";
        }
    }

    function verificarStatusContaAntesLogin(reDigitado, senhaDigitada) {
        const termo = reDigitado.trim().toLowerCase();
        const conta = contasCadastradas.find(c => c.re.toLowerCase() === termo || c.email.toLowerCase() === termo);
        
        if (conta) {
            if (conta.status === "pendente") {
                mostrarToast(`⚠️ Acesso Pendente: A conta (${conta.re}) aguarda aprovação de um Superior pelo site.`, "warning");
                return { ok: false, conta: conta };
            }
            if (conta.status === "rejeitado") {
                mostrarToast(`❌ Acesso Recusado: A solicitação de (${conta.re}) foi indeferida pelo Superior.`, "error");
                return { ok: false, conta: conta };
            }
            if (conta.senha && senhaDigitada && conta.senha !== senhaDigitada) {
                mostrarToast(`❌ Senha Incorreta. Verifique suas credenciais.`, "error");
                return { ok: false, conta: null };
            }
            return { ok: true, conta: conta };
        }
        return { ok: true, conta: null };
    }

    function atualizarEstatisticasAprovacao() {
        const total = contasCadastradas.length;
        const pendentes = contasCadastradas.filter(c => c.status === "pendente").length;
        const aprovadas = contasCadastradas.filter(c => c.status === "aprovado").length;
        const rejeitadas = contasCadastradas.filter(c => c.status === "rejeitado").length;

        if (document.getElementById("statTotalContas")) document.getElementById("statTotalContas").textContent = total;
        if (document.getElementById("statPendentesContas")) document.getElementById("statPendentesContas").textContent = pendentes;
        if (document.getElementById("statAprovadasContas")) document.getElementById("statAprovadasContas").textContent = aprovadas;
        if (document.getElementById("statRejeitadasContas")) document.getElementById("statRejeitadasContas").textContent = rejeitadas;

        atualizarBadgeContasPendentes();
    }

    function aprovarConta(id) {
        const index = contasCadastradas.findIndex(c => c.id === id);
        if (index !== -1) {
            contasCadastradas[index].status = "aprovado";
            localStorage.setItem("ellencoContasCadastradas", JSON.stringify(contasCadastradas));
            mostrarToast(`✅ Conta de ${contasCadastradas[index].nome} aprovada com sucesso! O funcionário já pode realizar o login.`, "success");
            renderizarTabelaAprovacoes();
            renderizarTabelaUsuarios();
        }
    }

    function aprovarTodasContas() {
        let cont = 0;
        contasCadastradas.forEach(c => {
            if (c.status === "pendente") {
                c.status = "aprovado";
                cont++;
            }
        });
        if (cont > 0) {
            localStorage.setItem("ellencoContasCadastradas", JSON.stringify(contasCadastradas));
            mostrarToast(`✅ ${cont} solicitações de conta foram aprovadas com sucesso!`, "success");
            renderizarTabelaAprovacoes();
            renderizarTabelaUsuarios();
        } else {
            mostrarToast("Não há solicitações pendentes para aprovar no momento.", "info");
        }
    }

    function abrirModalRejeicao(id) {
        const conta = contasCadastradas.find(c => c.id === id);
        if (!conta) return;
        document.getElementById("rejeicaoContaId").value = id;
        document.getElementById("motivoRejeicaoTexto").value = "";
        document.getElementById("subtituloRejeicaoConta").innerHTML = `Informe a justificativa de recusa para a solicitação de <strong>${conta.nome}</strong> (${conta.re}):`;
        document.getElementById("modalRejeicaoConta").classList.remove("hidden");
    }

    function fecharModalRejeicao() {
        document.getElementById("modalRejeicaoConta").classList.add("hidden");
    }

    function confirmarRejeicaoConta() {
        const id = parseInt(document.getElementById("rejeicaoContaId").value);
        const motivo = document.getElementById("motivoRejeicaoTexto").value.trim();

        if (!motivo) {
            mostrarToast("Por favor, preencha a justificativa do indeferimento.", "warning");
            return;
        }

        const index = contasCadastradas.findIndex(c => c.id === id);
        if (index !== -1) {
            contasCadastradas[index].status = "rejeitado";
            contasCadastradas[index].motivoRejeicao = motivo;
            localStorage.setItem("ellencoContasCadastradas", JSON.stringify(contasCadastradas));
            mostrarToast(`Solicitação de ${contasCadastradas[index].nome} rejeitada.`, "info");
            fecharModalRejeicao();
            renderizarTabelaAprovacoes();
            renderizarTabelaUsuarios();
        }
    }

    function atualizarBadgeContasPendentes() {
        const pendentes = contasCadastradas.filter(c => c.status === "pendente").length;
        const badge = document.getElementById("badgeContasPendentes");
        if (badge) {
            badge.textContent = pendentes;
            if (pendentes === 0) badge.classList.add("hidden");
            else badge.classList.remove("hidden");
        }
    }

    function renderizarTabelaUsuarios() {
        const tbody = document.getElementById("tabelaUsuariosGestao");
        if (!tbody) return;

        const termo = (document.getElementById("filtroUsuarioTexto")?.value || "").toLowerCase();
        const obraSel = document.getElementById("filtroUsuarioObra")?.value || "todas";
        const cargoSel = document.getElementById("filtroUsuarioCargo")?.value || "todos";

        tbody.innerHTML = "";

        let lista = contasCadastradas.filter(c => {
            if (termo && !c.nome.toLowerCase().includes(termo) && !c.re.toLowerCase().includes(termo) && !c.email.toLowerCase().includes(termo)) return false;
            if (obraSel !== "todas" && c.obra !== obraSel) return false;
            if (cargoSel !== "todos") {
                const info = obterRotorCargo(c.cargo);
                if (info.perfil !== cargoSel && c.cargo !== cargoSel) return false;
            }
            return true;
        });

        if (lista.length === 0) {
            tbody.innerHTML = `
                <tr>
                    <td colspan="7" class="p-8 text-center text-slate-400">
                        <i class="fa-solid fa-user-slash text-2xl mb-2"></i>
                        <p>Nenhum colaborador encontrado com os filtros selecionados.</p>
                    </td>
                </tr>
            `;
            return;
        }

        lista.forEach(c => {
            let badgeStatus = c.status === 'aprovado' 
                ? '<span class="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300 text-[10px] font-bold px-2 py-0.5 rounded-full">Ativo</span>'
                : (c.status === 'pendente' 
                    ? '<span class="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300 text-[10px] font-bold px-2 py-0.5 rounded-full">Pendente</span>'
                    : '<span class="bg-red-100 text-ellenco-red dark:bg-red-900/40 dark:text-red-300 text-[10px] font-bold px-2 py-0.5 rounded-full">Inativo</span>');

            tbody.innerHTML += `
                <tr class="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition-colors">
                    <td class="p-3.5 font-bold text-slate-900 dark:text-white flex items-center gap-2">
                        <div class="w-7 h-7 rounded-lg bg-ellenco-navy text-white flex items-center justify-center font-bold text-xs uppercase">${c.nome.charAt(0)}</div>
                        <span>${c.nome}</span>
                    </td>
                    <td class="p-3.5 font-mono font-bold text-slate-700 dark:text-slate-300">${c.re}</td>
                    <td class="p-3.5 text-slate-600 dark:text-slate-300">${c.email}</td>
                    <td class="p-3.5 font-semibold text-slate-800 dark:text-slate-200">${c.cargoLabel || c.cargo}</td>
                    <td class="p-3.5 text-slate-600 dark:text-slate-400">${c.obra}</td>
                    <td class="p-3.5 text-center">${badgeStatus}</td>
                    <td class="p-3.5 text-center">
                        <button onclick="abrirModalEditarUsuario(${c.id})" class="px-3 py-1 bg-slate-100 hover:bg-slate-200 dark:bg-slate-800 dark:hover:bg-slate-700 text-slate-700 dark:text-slate-200 font-bold text-[11px] rounded-lg transition-all flex items-center justify-center gap-1 mx-auto">
                            <i class="fa-solid fa-pen-to-square text-ellenco-blue"></i> Editar
                        </button>
                    </td>
                </tr>
            `;
        });
    }

    function abrirModalEditarUsuario(id) {
        const conta = contasCadastradas.find(c => c.id === id);
        if (!conta) return;

        document.getElementById("editUserId").value = id;
        document.getElementById("editUserNome").value = conta.nome;
        document.getElementById("editUserRE").value = conta.re;
        document.getElementById("editUserEmail").value = conta.email;
        document.getElementById("editUserCargo").value = conta.cargo || "encarregado";
        document.getElementById("editUserObra").value = conta.obra || "SP-270 Rodovia Raposo Tavares";

        document.getElementById("modalEditarUsuario").classList.remove("hidden");

    }

    function salvarEdicaoUsuario(e) {
        e.preventDefault();
        const id = parseInt(document.getElementById("editUserId").value);
        const index = contasCadastradas.findIndex(c => c.id === id);

        if (index !== -1) {
            const cargoVal = document.getElementById("editUserCargo").value;
            const infoCargo = obterRotorCargo(cargoVal);

            contasCadastradas[index].nome = document.getElementById("editUserNome").value.trim();
            contasCadastradas[index].email = document.getElementById("editUserEmail").value.trim();
            contasCadastradas[index].cargo = cargoVal;
            contasCadastradas[index].cargoLabel = infoCargo.label;
            contasCadastradas[index].nivelCargo = infoCargo.nivel;
            contasCadastradas[index].obra = document.getElementById("editUserObra").value;
            contasCadastradas[index].status = document.getElementById("editUserStatus").value;

            const novaSenha = document.getElementById("editUserNovaSenha").value.trim();
            if (novaSenha) contasCadastradas[index].senha = novaSenha;

            localStorage.setItem("ellencoContasCadastradas", JSON.stringify(contasCadastradas));
            mostrarToast(`✅ Perfil de ${contasCadastradas[index].nome} atualizado com sucesso!`, "success");
            fecharModalEditarUsuario();
            renderizarTabelaAprovacoes();
            renderizarTabelaUsuarios();
        }
    }

            "Nome Completo": c.nome,

            "E-mail Corporativo": c.email,
            "Telefone": c.telefone || "",
            "Cargo": c.cargoLabel || c.cargo,
            "Nível Hierárquico": c.nivelCargo || 1,
            "Obra": c.obra,
            "Status no Sistema": c.status.toUpperCase(),
            "Data de Solicitação": c.dataSolicitacao

        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, ws, "Colaboradores Ellenco");
        XLSX.writeFile(wb, `Colaboradores_Ellenco_${new Date().toISOString().split("T")[0]}.xlsx`);
        mostrarToast("📊 Planilha Excel de colaboradores exportada com sucesso!", "success");
    }

    function efetuarLogin(e) {
        e.preventDefault();
        
        const reInput = document.getElementById("loginRE").value;
        const senhaInput = document.getElementById("loginSenha").value;
        
        const valid = verificarStatusContaAntesLogin(reInput, senhaInput);
        if (!valid.ok) return;

        const email = document.getElementById("loginEmail") ? document.getElementById("loginEmail").value : (valid.conta ? valid.conta.email : reInput);
        const perfil = document.getElementById("loginPerfil").value;
        const codigoObra = document.getElementById("loginCodigoObra").value || (valid.conta ? valid.conta.obra : "SP-270");
        
        let nomeUsuario = perfil === 'operacional' ? "Carlos Silva (Encarregado)" : "Eng. Ricardo Viana (Diretoria)";
        let reUser = reInput;

        if (valid.conta) {
            nomeUsuario = valid.conta.nome + " (" + (valid.conta.cargoLabel || valid.conta.cargo) + ")";
            reUser = valid.conta.re;
        }
        
        usuarioSessao = {
            logado: true,
            nome: nomeUsuario,
            email: email,
            re: reUser,
            perfil: perfil,
            obra: codigoObra
        };
        
        localStorage.setItem("ellencoSessao", JSON.stringify(usuarioSessao));
        mostrarToast(`Bem-vindo, ${usuarioSessao.nome}!`, "success");
        verificarAutenticacao();
        
        const pendentes = contasCadastradas.filter(c => c.status === "pendente").length;
        if ((perfil === 'gestor' || perfil === 'engenheiro') && pendentes > 0) {
            mostrarToast(`🔔 Notificação de Superior: Existem ${pendentes} cadastros de funcionários aguardando sua aprovação na aba Aprovações!`, "info");
        }
    }

    function inicializarTema() {
        const temaSalvo = localStorage.getItem("ellencoTheme") || "light";
        if (temaSalvo === "dark") {
            document.documentElement.classList.add("dark");
            atualizarIconeTema(true);
        } else {
            document.documentElement.classList.remove("dark");
            atualizarIconeTema(false);
        }
    }

    function toggleDarkMode() {
        const isDark = document.documentElement.classList.toggle("dark");
        localStorage.setItem("ellencoTheme", isDark ? "dark" : "light");
        atualizarIconeTema(isDark);
        mostrarToast(`Modo ${isDark ? 'Escuro' : 'Claro'} ativado`, "info");
    }

    function mostrarPagina(paginaId, btnElement = null) {
        document.querySelectorAll(".pagina").forEach(sec => sec.classList.add("hidden"));

        const alvo = document.getElementById("pagina-" + paginaId);
        if (alvo) {
            alvo.classList.remove("hidden");
        }

        document.querySelectorAll(".menu-item").forEach(item => item.classList.remove("active"));
        if (btnElement) {
            btnElement.classList.add("active");
        }

        const titulos = {
            dashboard: ["Dashboard & KPIs", "Visão geral e comparativo imediato da produção"],
            apontamento: ["Novo Apontamento Digital", "Registro e conferência automatizada de produção"],
            historico: ["Histórico & Auditoria", "Consulta e exportação dos registros validados"],
            equipes: ["Equipes de Campo", "Gestão de efetivo e métricas por equipe"],
            obras: ["Obras & Frentes de Trabalho", "Status físico e infraestrutura em andamento"],
            equipamentos: ["Equipamentos & Frota", "Controle de maquinário e tempo de operação"],
            relatorios: ["Relatórios Executivos", "Emissão de relatórios oficiais para diretoria"],
            aprovacoes: ["Aprovação de Contas", "Central de análise e concessão de acessos por cargo superior"],
            usuarios: ["Gestão de Colaboradores", "Administração de perfis, senhas e auditoria de usuários"]
        };

        if (titulos[paginaId]) {
            document.getElementById("tituloPagina").textContent = titulos[paginaId][0];
            document.getElementById("subtituloPagina").textContent = titulos[paginaId][1];
        }

        if (paginaId === "historico") renderizarHistorico();
        if (paginaId === "aprovacoes") renderizarTabelaAprovacoes();
        if (paginaId === "usuarios") renderizarTabelaUsuarios();

        // Close sidebar on mobile

    function calcularDesvio() {
        const planejado = parseFloat(document.getElementById("planejado").value);
        const realizado = parseFloat(document.getElementById("realizado").value);
        const horas = parseFloat(document.getElementById("horas").value) || 8;

        const container = document.getElementById("resultadoCalculo");
        const box = document.getElementById("boxResultado");
        const icone = document.getElementById("iconeResultado");
        const texto = document.getElementById("textoResultado");
        const detalhe = document.getElementById("detalheResultado");
        const avisoTypo = document.getElementById("avisoAlertaTypo");

        if (isNaN(planejado) || isNaN(realizado) || planejado <= 0) {
            container.classList.add("hidden");
            return;
        }

        container.classList.remove("hidden");

        const desvio = ((realizado - planejado) / planejado) * 100;
        const eficiencia = (realizado / planejado) * 100;
        const prod = horas > 0 ? (realizado / horas).toFixed(1) : 0;
        const u = document.getElementById("unidade").value;

        document.getElementById("resEficiencia").textContent = eficiencia.toFixed(1) + "%";
        document.getElementById("resDesvio").textContent = (desvio >= 0 ? "+" : "") + desvio.toFixed(1) + "%";
        document.getElementById("resProdutividade").textContent = prod + " " + u.substring(0,3) + "/h";

        // Anti-Typo Safety Check
        if (realizado > planejado * 3) {
            avisoTypo.classList.remove("hidden");
        } else {
            avisoTypo.classList.add("hidden");
        }

        if (desvio >= 0) {
            box.className = "rounded-xl p-4 border flex flex-col md:flex-row justify-between items-start md:items-center gap-4 bg-emerald-50/70 border-emerald-200 shadow-sm";
            icone.className = "w-12 h-12 rounded-xl flex items-center justify-center text-xl shrink-0 bg-emerald-500 text-white shadow-md shadow-emerald-500/20";
            icone.innerHTML = '<i class="fa-solid fa-circle-check"></i>';
            texto.textContent = "Meta Atingida com Sucesso!";
            texto.className = "text-base font-extrabold text-emerald-800";
            detalhe.textContent = `Produção +${desvio.toFixed(1)}% acima do planejado. Desempenho excelente da equipe.`;
        } else if (desvio >= -10) {
            box.className = "rounded-xl p-4 border flex flex-col md:flex-row justify-between items-start md:items-center gap-4 bg-amber-50/70 border-amber-200 shadow-sm";
            icone.className = "w-12 h-12 rounded-xl flex items-center justify-center text-xl shrink-0 bg-amber-500 text-white shadow-md shadow-amber-500/20";
            icone.innerHTML = '<i class="fa-solid fa-triangle-exclamation"></i>';
            texto.textContent = "Atenção: Pequeno Desvio Identificado";
            texto.className = "text-base font-extrabold text-amber-800";
            detalhe.textContent = `Produção ${desvio.toFixed(1)}% abaixo da meta programada. Acompanhar ritmo.`;
        } else {
            box.className = "rounded-xl p-4 border flex flex-col md:flex-row justify-between items-start md:items-center gap-4 bg-red-50/70 border-red-200 shadow-sm";
            icone.className = "w-12 h-12 rounded-xl flex items-center justify-center text-xl shrink-0 bg-ellenco-red text-white shadow-md shadow-red-500/20";
            icone.innerHTML = '<i class="fa-solid fa-circle-xmark"></i>';
            texto.textContent = "Alerta Crítico: Produção Abaixo da Meta";
            texto.className = "text-base font-extrabold text-red-900";
            detalhe.textContent = `Desvio significativo de ${desvio.toFixed(1)}%. Obrigatório justificar paralisação/ocorrência.`;
        }
    }

        const realizado = parseFloat(document.getElementById("realizado").value);

    // 6. DASHBOARD CALCULATIONS & REFRESH
    function atualizarDashboard() {
        const pFiltro = document.getElementById("filtroGlobalPeriodo").value;
        const oFiltro = document.getElementById("filtroGlobalObra").value;

        let dadosFiltrados = apontamentos.filter(item => {
            let okObra = (oFiltro === "todas" || item.obra === oFiltro);
            let okPeriodo = true;
            const hojeStr = new Date().toISOString().split("T")[0];

            if (pFiltro === "hoje") {
                okPeriodo = (item.data === hojeStr);
            } else if (pFiltro === "semana") {
                const diffTime = Math.abs(new Date() - new Date(item.data));
                const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
                okPeriodo = (diffDays <= 7);
            }
            return okObra && okPeriodo;
        });

        if (dadosFiltrados.length === 0) dadosFiltrados = apontamentos;

        const totalMeta = dadosFiltrados.reduce((acc, i) => acc + i.planejado, 0);
        const totalRealizado = dadosFiltrados.reduce((acc, i) => acc + i.realizado, 0);
        const totalHoras = dadosFiltrados.reduce((acc, i) => acc + (i.horas || 8), 0);

        const efiGlobal = totalMeta > 0 ? (totalRealizado / totalMeta) * 100 : 0;
        const desvioGlobal = efiGlobal - 100;
        const prodMedia = totalHoras > 0 ? (totalRealizado / totalHoras) : 0;

        document.getElementById("cardMeta").textContent = totalMeta.toLocaleString("pt-BR");
        document.getElementById("cardRealizado").textContent = totalRealizado.toLocaleString("pt-BR");
        document.getElementById("cardEficiencia").textContent = efiGlobal.toFixed(1) + "%";

        const elDesvio = document.getElementById("cardDesvio");
        elDesvio.textContent = (desvioGlobal >= 0 ? "+" : "") + desvioGlobal.toFixed(1) + "%";
        elDesvio.className = "text-2xl font-black " + (desvioGlobal >= 0 ? "text-emerald-600" : "text-ellenco-red");

        document.getElementById("cardProdutividade").textContent = prodMedia.toFixed(1) + " /h";

        // Count Team Performance Breakdown
        let acima = 0, naMeta = 0, abaixo = 0;
        dadosFiltrados.forEach(i => {
            const ef = (i.realizado / i.planejado) * 100;
            if (ef >= 100) acima++;
            else if (ef >= 90) naMeta++;
            else abaixo++;
        });

        document.getElementById("statAcimaMeta").textContent = acima;
        document.getElementById("statNaMeta").textContent = naMeta;
        document.getElementById("statAbaixoMeta").textContent = abaixo;

        // Render Dashboard Team Table
        renderizarTabelaEquipesDashboard(dadosFiltrados);

        // Update Charts
        atualizarGraficos(dadosFiltrados, acima, naMeta, abaixo);
    }

    function aplicarFiltrosGlobais() {
        atualizarDashboard();
    }

    function renderizarTabelaEquipesDashboard(dados) {
        const tbody = document.getElementById("tabelaEquipesDashboard");
        tbody.innerHTML = "";

        const equipesAgrupadas = {};
        dados.forEach(i => {
            if (!equipesAgrupadas[i.equipe]) {
                equipesAgrupadas[i.equipe] = {
                    equipe: i.equipe,
                    atividade: i.atividade,
                    obra: i.obra,
                    planejado: 0,
                    realizado: 0
                };
            }
            equipesAgrupadas[i.equipe].planejado += i.planejado;
            equipesAgrupadas[i.equipe].realizado += i.realizado;
        });

        Object.values(equipesAgrupadas).forEach(eq => {
            const ef = (eq.realizado / eq.planejado) * 100;
            let statusBadge = '';
            let progressBg = '';

            if (ef >= 100) {
                statusBadge = '<span class="bg-emerald-100 text-emerald-800 text-[11px] font-bold px-2.5 py-1 rounded-full">Acima da Meta</span>';
                progressBg = 'bg-emerald-500';
            } else if (ef >= 90) {
                statusBadge = '<span class="bg-amber-100 text-amber-800 text-[11px] font-bold px-2.5 py-1 rounded-full">Na Meta</span>';
                progressBg = 'bg-amber-500';
            } else {
                statusBadge = '<span class="bg-red-100 text-ellenco-red text-[11px] font-bold px-2.5 py-1 rounded-full">Abaixo da Meta</span>';
                progressBg = 'bg-ellenco-red';
            }

            tbody.innerHTML += `
                <tr class="hover:bg-slate-50 transition-colors">
                    <td class="py-3 px-4 font-bold text-slate-900">${eq.equipe}</td>
                    <td class="py-3 px-4 text-slate-600">${eq.atividade}</td>
                    <td class="py-3 px-4 text-slate-600">${eq.obra}</td>
                    <td class="py-3 px-4 text-right font-semibold text-slate-700">${eq.planejado.toLocaleString("pt-BR")}</td>
                    <td class="py-3 px-4 text-right font-bold text-slate-900">${eq.realizado.toLocaleString("pt-BR")}</td>
                    <td class="py-3 px-4">
                        <div class="flex items-center gap-2">
                            <div class="w-16 bg-slate-200 h-2 rounded-full overflow-hidden">
                                <div class="h-2 ${progressBg} rounded-full" style="width: ${Math.min(ef, 100)}%"></div>
                            </div>
                            <span class="font-bold text-xs">${ef.toFixed(0)}%</span>
                        </div>
                    </td>
                    <td class="py-3 px-4 text-center">${statusBadge}</td>
                </tr>
            `;
        });
    }

    // 8. HISTÓRICO & AUDITORIA RENDERER
    function renderizarHistorico() {
        const tbody = document.getElementById("tabelaHistorico");
        const filtroTexto = document.getElementById("filtroHistorico").value.toLowerCase();
        const filtroStatus = document.getElementById("filtroStatusHistorico").value;
        const filtroObra = document.getElementById("filtroObraHistorico").value;

        let lista = apontamentos.filter(i => {
            const okTexto = i.equipe.toLowerCase().includes(filtroTexto) || i.obra.toLowerCase().includes(filtroTexto) || i.atividade.toLowerCase().includes(filtroTexto);
            const okObra = (filtroObra === "todas" || i.obra === filtroObra);

            let okStatus = true;
            const desv = i.desvio || 0;
            if (filtroStatus === "acima") okStatus = (desv >= 0);
            else if (filtroStatus === "atencao") okStatus = (desv >= -10 && desv < 0);
            else if (filtroStatus === "critico") okStatus = (desv < -10);

            return okTexto && okObra && okStatus;
        });

        tbody.innerHTML = "";

        if (lista.length === 0) {
            tbody.innerHTML = `
                <tr>
                    <td colspan="9" class="p-8 text-center text-slate-400">
                        <i class="fa-solid fa-folder-open text-3xl mb-2"></i>
                        <p>Nenhum apontamento encontrado para os filtros selecionados.</p>
                    </td>
                </tr>
            `;
            document.getElementById("contadorRegistrosHistorico").textContent = "Exibindo 0 registros";
            return;
        }

        document.getElementById("contadorRegistrosHistorico").textContent = `Exibindo ${lista.length} registros`;

        lista.slice().reverse().forEach(item => {
            const dataFmt = item.data ? item.data.split("-").reverse().join("/") : "-";
            const desv = item.desvio || 0;

            let badgeStatus = '';
            if (desv >= 0) {
                badgeStatus = '<span class="bg-emerald-100 text-emerald-800 text-[11px] font-bold px-2.5 py-0.5 rounded-full">Meta Atingida</span>';
            } else if (desv >= -10) {
                badgeStatus = '<span class="bg-amber-100 text-amber-800 text-[11px] font-bold px-2.5 py-0.5 rounded-full">Atenção</span>';
            } else {
                badgeStatus = '<span class="bg-red-100 text-ellenco-red text-[11px] font-bold px-2.5 py-0.5 rounded-full">Crítico</span>';
            }

            tbody.innerHTML += `
                <tr class="hover:bg-slate-50 transition-colors">
                    <td class="p-3 font-semibold text-slate-700">${dataFmt}</td>
                    <td class="p-3 font-bold text-slate-900">${item.obra}</td>
                    <td class="p-3 text-slate-700">${item.equipe}</td>
                    <td class="p-3 text-slate-600">${item.atividade}</td>
                    <td class="p-3 text-right font-semibold text-slate-600">${item.planejado.toLocaleString("pt-BR")} ${item.unidade}</td>
                    <td class="p-3 text-right font-bold text-slate-900">${item.realizado.toLocaleString("pt-BR")} ${item.unidade}</td>
                    <td class="p-3 text-right font-bold ${desv >= 0 ? 'text-emerald-600' : 'text-ellenco-red'}">
                        ${desv >= 0 ? '+' : ''}${desv.toFixed(1)}%
                    </td>
                    <td class="p-3 text-center">${badgeStatus}</td>
                    <td class="p-3 text-center no-print">
                        <button onclick="excluirApontamento(${item.id})" class="text-slate-400 hover:text-ellenco-red p-1"><i class="fa-solid fa-trash"></i></button>
                    </td>
                </tr>
            `;
        });
    }

        if (confirm("Deseja realmente remover este registro de apontamento?")) {
            apontamentos = apontamentos.filter(i => i.id !== id);
            localStorage.setItem("ellencoApontamentos", JSON.stringify(apontamentos));
            renderizarHistorico();
            atualizarDashboard();
            mostrarToast("Registro excluído com sucesso.", "info");
        }
    }

    function limparHistorico() {
        if (confirm("ATENÇÃO: Deseja apagar TODO o histórico de apontamentos salvos?")) {
            apontamentos = [];
            localStorage.removeItem("ellencoApontamentos");
            renderizarHistorico();
            atualizarDashboard();
            mostrarToast("Histórico limpo.", "info");
        }
    }

                            <i class="fa-solid fa-location-dot text-ellenco-red"></i> ${ob.local} • ${ob.eng}
                        </p>
                        <div class="space-y-1.5 text-xs">

                                <span>Progresso Físico</span>

                    </div>
                </div>
            `;

        e.preventDefault();
        const nova = {
            id: Date.now(),
            nome: document.getElementById("novaEquipeNome").value,
            encarregado: document.getElementById("novaEquipeEncarregado").value,
            atividade: document.getElementById("novaEquipeAtividade").value,

    // 11. TOAST NOTIFICATION SYSTEM
    function mostrarToast(mensagem, tipo = "info") {
        const container = document.getElementById("toastContainer");
        const toast = document.createElement("div");

        const cores = {
            success: "bg-emerald-600 text-white",
            error: "bg-ellenco-red text-white",
            warning: "bg-amber-500 text-white",
            info: "bg-ellenco-navy text-white"
        };

        toast.className = `p-4 rounded-xl shadow-xl text-xs font-bold flex items-center gap-3 transition-all duration-300 pointer-events-auto ${cores[tipo] || cores.info}`;
        toast.innerHTML = `<i class="fa-solid fa-circle-info text-base"></i> <span>${mensagem}</span>`;

        container.appendChild(toast);

        setTimeout(() => {
            toast.style.opacity = "0";
            setTimeout(() => toast.remove(), 300);
        }, 3500);
    }
